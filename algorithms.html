  12. Singly Linked List

    1. Start
    2. Repeat until user choose to exit:
      1. Display Menu: Insert at Beginning, Insert at Middle, Insert at End, Delete at Beginning, Delete at Middle, Delete at End, Display, Exit
      2. Read choice
      3. Perform selected operation
    3. Insert at Beginning
      1. Read val
      2. newNode = malloc(sizeof(struct Node))
      3. newNode.data = val
      4. newNode.next = NULL
      5. if (head == NULL) then
        1. head = newNode
        2. exit
      6. else
        1. newNode.next = head
        2. head = newNode
      7. endif
    4. Insert at Middle
      1. Read val, key
      2. newNode = malloc(sizeof(struct Node))
      3. newNode.data = val
      4. newnode.next = NULL
      5. if (head == NULL) then
        1. print ("List is empty")
        2. exit
      6. endif
      7. temp = head
      8. while (temp != NULL and temp.data != key)
        1. temp = temp.next
      9. endwhile
      10. if (temp == NULL) then
        1. print ("Key not found!")
        2. exit
      11. endif
      12. newNode.next = temp.next
      13. temp.next = newNode
    5. Insert at End
      1. Read val
      2. newNode = malloc(sizeof(struct Node))
      3. newNode.data = val
      4. newNode.next = NULL
      5. if (head == NULL) then
        1. head = newNode
        2. exit
      6. endif
      7. temp = head
      8. while (temp.next != NULL)
        1. temp = temp.next
      9. endwhile
      10. temp.next = newNode
    6. Delete at Beginning
      1. if (head == NULL) then
        1. print ("List is empty")
        2. exit
      2. endif
      3. temp = head
      4. head = head.next
      5. print temp.data
      6. free(temp)
    7. Delete at Middle
      1. if (head == NULL) then
        1. print ("List is empty")
        2. exit
      2. endif
      3. Read key
      4. if (head.data == key) then
        1. temp = head
        2. head = head.next
        3. print temp.data
        4. free(temp)
        5. exit
      5. endif
      6. temp = head
      7. while (temp.next != NULL and temp.next.data != key)
        1. temp = temp.next
      8. endwhile
      9. if (temp.next == NULL) then
        1. print ("Key not found!")
        2. exit
      10. endif
      11. delNode = temp.next
      12. temp.next = delNode.next
      13. print delNode.data
      14. free(delNode)
    8. Delete at End
      1. if (head == NULL) then
        1. print ("List is empty")
        2. exit
      2. endif
      3. if (head.next == NULL) then
        1. temp = head
        2. head = NULL
        3. print temp.data
        4. free(temp)
        5. exit
      4. endif
      5. temp = head
      6. while (temp.next.next != NULL)
        1. temp = temp.next
      7. endwhile
      8. delNode = temp.next
      9. temp.next = NULL
      10. print delNode.data
      11. free(delNode)
    9. Display
      1. if (head == NULL) then
        1. print ("List is empty")
        2. exit
      2. else
        1. ptr = head
        2. while (ptr != NULL)
          1. print ptr.data
          2. ptr = ptr.next
        3. endwhile
      3. endif
    10. Stop

  14. Stack Using a Linked List

    3. Push
      1. Read item
      2. newnode = malloc(sizeof(struct node))
      3. newnode.data = item
      4. newnode.next = NULL
      5. if (TOP == NULL) then
        1. TOP = newnode
        2. exit
      6. else
        1. newnode.next = TOP
        2. TOP = newnode
      7. endif
    4. Pop
      1. if (TOP == NULL) then
        1. print ("Stack Empty / Underflow")
        2. exit
      2. else
        1. temp = TOP
        2. TOP = TOP.next
        3. print temp.data
        4. free(temp)
      3. endif
    5. Display
      1. if (TOP == NULL) then
        1. print ("Stack Empty")
        2. exit
      2. else
        1. ptr = TOP
        2. while (ptr != NULL)
          1. print ptr.data
          2. ptr = ptr.next
      3. endwhile
      4. endif
    6. Stop

    3. Enqueue

      3. newnode.data = item
      4. newnode.next = NULL
      5. if (FRONT == NULL and REAR == NULL) then
        1. FRONT = newnode
        2. REAR = newnode
        3. exit
      6. else
        1. REAR.next = newnode
        2. REAR = newnode
        3. endif
    4. Dequeue
      1. if (FRONT == NULL) then
        1. print ("Queue Empty / Underflow")
        2. exit
      2. else
        1. temp = FRONT
        2. FRONT = FRONT.next
        3. print temp.data
        4. free(temp)
        5. if (FRONT == NULL) then
          1. REAR = NULL
        6. endif
      3. endif
    5. Display
