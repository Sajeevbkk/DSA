  14. Stack Using a Linked List

    1. Start
    2. Repeat until user choose to exit:
      1. Display Menu: Push, Pop, Display, exit
      2. Read choice
      3. Perform selected operation
    3. Push
      1. Read item
      2. newnode = malloc(sizeof(struct node))
      3. newnode.data = item
      4. newnode.next = NULL
      5. if (TOP == NULL) then
        1. TOP = newnode
        2. exit
      6. else
        1. newnode.next = TOP
        2. TOP = newnode
      7. endif
    4. Pop
      1. if (TOP == NULL) then
        1. print ("Stack Empty / Underflow")
        2. exit
      2. else
        1. temp = TOP
        2. TOP = TOP.next
        3. print temp.data
        4. free(temp)
      3. endif
    5. Display
      1. if (TOP == NULL) then
        1. print ("Stack Empty")
        2. exit
      2. else
        1. ptr = TOP
        2. while (ptr != NULL)
          1. print ptr.data
          2. ptr = ptr.next
      3. endwhile
      4. endif
    6. Stop

  15. Queue Using a Linked List

    1. Start
    2. Repeat until user choose to exit:
      1. Display Menu: Enqueu, Dequeue, Display, exit
      2. Read choice
      3. Perform selected operation
    3. Enqueue
      1. Read item
      2. newnode = malloc(sizeof(struct node))
      3. newnode.data = item
      4. newnode.next = NULL
      5. if (FRONT == NULL and REAR == NULL) then
        1. FRONT = newnode
        2. REAR = newnode
        3. exit
      6. else
        1. REAR.next = newnode
        2. REAR = newnode
        3. endif
    4. Dequeue
      1. if (FRONT == NULL) then
        1. print ("Queue Empty / Underflow")
        2. exit
      2. else
        1. temp = FRONT
        2. FRONT = FRONT.next
        3. print temp.data
        4. free(temp)
        5. if (FRONT == NULL) then
          1. REAR = NULL
        6. endif
      3. endif
    5. Display
      1. if (FRONT == NULL) then
        1. print ("Queue Empty")
        2. exit
      2. else
        1. ptr = FRONT
        2. while (ptr != NULL)
          1. print ptr.data
          2. ptr = ptr.next
        3. endwhile
      3. endif
    6. Stop
